  COMMAND ${CMAKE_COMMAND} -E copy_if_different

    "${RESOURCE_DIR}/f3dex2.json"
)

# Embed the RSP overlay descriptors into the binary as a compiled-in fallback, used
# at runtime when the JSON files can't be located on disk (e.g. an installed build
# where they weren't shipped next to the executable). Generated at configure time
# from the same JSON files; CMAKE_CONFIGURE_DEPENDS re-runs configure (regenerating
# the header) whenever a JSON changes, so the embedded copy never goes stale.
set(EMBED_RSPQ_JSON "${CMAKE_CURRENT_SOURCE_DIR}/ui/rsp-overlays/rspq-libdragon.json")
set(EMBED_F3D_JSON  "${CMAKE_CURRENT_SOURCE_DIR}/ui/rsp-overlays/f3dex2.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${EMBED_RSPQ_JSON}" "${EMBED_F3D_JSON}")
file(READ "${EMBED_RSPQ_JSON}" EMBED_RSPQ_CONTENT)
file(READ "${EMBED_F3D_JSON}"  EMBED_F3D_CONTENT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/embedded-overlays.hpp"
"// Auto-generated by CMake from ui/rsp-overlays/*.json. Do not edit.\n"
"#pragma once\n"
"namespace ares::ui {\n"
"inline const char embeddedRspqJson[] = R\"JSONEMBED(${EMBED_RSPQ_CONTENT})JSONEMBED\";\n"
"inline const char embeddedF3dJson[]  = R\"JSONEMBED(${EMBED_F3D_CONTENT})JSONEMBED\";\n"
"}  // namespace ares::ui\n")
target_include_directories(desktop-ui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
